逆波兰表达式
12 34 + 32 /  22 11 + -
```c++
#include <iostream>
#include <vector>
#include <string>
#include <sstream>
#include <cctype>
using namespace std;

// 按空格切分表达式
vector<string> split(const string& s) {
    vector<string> toks;
    stringstream ss(s);
    string t;
    while (ss >> t) toks.push_back(t);
    return toks;
}

// 从右往左递归求值；pos 是共享下标，每取一个 token 就左移一位
double eval(const vector<string>& toks, int& pos) {
    const string& t = toks[pos--];

    // 是数字（含负数）——叶子节点，直接返回
    if (isdigit(t[0]) || (t.size() > 1 && t[0] == '-'))
        return stod(t);

    // 是运算符——先递归右操作数，再递归左操作数
    double right = eval(toks, pos);   // 更靠右的那个 = 右操作数
    double left  = eval(toks, pos);   // 再往左 = 左操作数

    switch (t[0]) {
        case '+': return left + right;
        case '-': return left - right;
        case '*': return left * right;
        case '/': return left / right;
        default:  return 0;
    }
}

double bolan(const string& expr) {
    vector<string> toks = split(expr);
    int pos = (int)toks.size() - 1;
    return eval(toks, pos);
}

int main() {
    cout << "support + - * /" << endl;
    cout << "输入一行后缀表达式（空格分隔），例如 12 34 +" << endl;
    while (true) {
        cout << "> ";
        string line;
        getline(cin, line);
        if (line.empty() || line == "exit") break;
        cout << "= " << bolan(line) << endl << endl;
    }
    return 0;
}
```
