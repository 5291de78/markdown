```asm
	.file	"2binary.c"
	.text
	.section .rdata,"dr"
.LC0:
	.ascii "Hello World\0"
	.text
	.globl	main
	.def	main;	.scl	2;	.type	32;	.endef
	.seh_proc	main
main:		
;当你读到这个的时候，小灯可能还在疯狂补AT&T的语法
	pushq	%rbp		
	;把**旧的基指针 rbp**压入栈，保存上一层函数栈底。
	.seh_pushreg	%rbp  
	;保存 rbp
	movq	%rsp, %rbp		
	;rbp 是被压栈保存的寄存器，异常处理时用来展开栈
	.seh_setframe	%rbp, 0 
	;把当前栈指针 rsp 赋值给 rbp，**建立本函数的栈帧**。rbp 固定为本函数栈底。
	subq	$32, %rsp
	;伪指令，SEH 标记：栈帧基址用 rbp，偏移 0
	.seh_stackalloc	32
	;栈向下增长，**开辟 32 字节栈空间（本函数的局部栈）**。Windows x64 要求函数调用时预留**影子空间 (shadow space)**，这里 32 字节就是分配栈 + 影子空间。
	.seh_endprologue
	;伪指令，标记**函数序言 (prologue) 结束**
	call	__main
	;调用 `__main`，MSYS2/gcc 的初始化函数，**C++ 全局变量 / 静态变量初始化**；纯 C 程序也会生成这个调用。
	leaq	.LC0(%rip), %rax
	movq	%rax, %rcx
	;rcx = rax
	call	printf
	;调用库函数 printf，打印字符串。
	movl	$0, %eax
	;eax = 0
	addq	$32, %rsp
	;rsp = rsp +32
	popq	%rbp
	;弹出栈里保存的旧 rbp，恢复上层函数的栈底
	ret
	;返回指令，弹出栈上返回地址，回到调用 main 的地方。
	.seh_endproc
	.def	__main;	.scl	2;	.type	32;	.endef
	;声明`printf`是外部函数，链接时去 lib 里找 printf 实现。
	.ident	"GCC: (Rev3, Built by MSYS2 project) 16.2.0"
	.def	printf;	.scl	2;	.type	32;	.endef
```
