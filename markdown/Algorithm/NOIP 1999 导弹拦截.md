NOIP 1999 导弹拦截

O(n^2)

```c
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    vector<int> a;
    int x;
    while (cin >> x) a.push_back(x);   // 读一行到 EOF
    int n = a.size();

    // 第一问：最长不上升子序列（a[j] >= a[i] 才能接上）
    vector<int> dp1(n, 1);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < i; j++)
            if (a[j] >= a[i]) dp1[i] = max(dp1[i], dp1[j] + 1);
    	//以第i个导弹结尾，前面 

    // 第二问：最长严格上升子序列（a[j] < a[i]）
    vector<int> dp2(n, 1);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < i; j++)
            if (a[j] < a[i]) dp2[i] = max(dp2[i], dp2[j] + 1);

    cout << *max_element(dp1.begin(), dp1.end()) << '\n'
         << *max_element(dp2.begin(), dp2.end()) << '\n';
    return 0;
}
```

O(n log n) 动态规划 + 贪心优化

```c
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    vector<int> a;
    int x;
    while (cin >> x) a.push_back(x);

    // 第一问：最长不上升子序列 = 对 -a 求「最长不降子序列」
    vector<int> d1;
    for (int v : a) {
        int y = -v;
        auto it = upper_bound(d1.begin(), d1.end(), y); // 不降 → upper_bound
        if (it == d1.end()) d1.push_back(y);
        else *it = y;
    }

    // 第二问：最长严格上升子序列
    vector<int> d2;
    for (int v : a) {
        auto it = lower_bound(d2.begin(), d2.end(), v); // 严格增 → lower_bound
        if (it == d2.end()) d2.push_back(v);
        else *it = v;
    }

    cout << d1.size() << '\n' << d2.size() << '\n';
    return 0;
}
```
